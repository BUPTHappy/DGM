\begin{algorithm}[t]
\caption{Dynamic Generative Modeling Optimization (DGMO)}
\label{alg:dgmo}
\begin{algorithmic}[1]
\Require Dataset $\mathcal{D}$, model $F_\theta$, hyperparameters $\mathbf{x} = \{\lambda, n, T, \ldots\}$, mode $\mathcal{M}$, epochs $E$, BO epochs $(E_s, \Delta E)$
\Ensure Trained $F_\theta$ and optimized $\mathbf{x}^*$

\Repeat
    \State \textbf{Optimize Model Parameters} $\theta$:
    \State \quad Sample $(x, z) \sim \mathcal{D}$, compute UCGM loss $\mathcal{L}_t(\theta, \mathbf{x})$ with $\lambda \in \mathbf{x}$
    \State \quad Update: $\theta \gets \theta - \eta \nabla_\theta \mathcal{L}_t(\theta, \mathbf{x})$
    
    \State \textbf{Optimize Hyperparameters} $\mathbf{x}$ (every $\Delta E$ epochs after $E_s$):
    \If{epoch $e \geq E_s$ and $(e - E_s) \bmod \Delta E = 0$}
        \State \quad $\mathbf{x}^* \gets \arg\max_{\mathbf{x} \in \mathcal{A}(\mathcal{M})} f(F_\theta, \mathbf{x}, \mathcal{D}_{\text{val}})$ via Bayesian optimization
        \State \quad Update: $\mathbf{x} \gets \mathbf{x}^*$
    \EndIf
\Until{Convergence}

\State Return $F_\theta$ and $\mathbf{x}^*$
\end{algorithmic}
\end{algorithm}
